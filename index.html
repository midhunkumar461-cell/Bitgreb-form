export default { async fetch(request, env) { if (request.method !== "POST") { return new Response("Method Not Allowed", { status: 405 }); } try { const data = await request.formData(); const name = data.get("name"); const email = data.get("email"); const message = data.get("message"); if (!email) { return new Response("Email is required", { status: 400 }); } const emailContent = ` New BitGreb form submission Name: ${name || "Not provided"} Email: ${email} Message: ${message || "No message"} `; await env.EMAIL.send({ from: "noreply@bitgreb.com", to: "YOUR_EMAIL@example.com", subject: "New BitGreb Form Submission", text: emailContent }); return new Response("Success", { status: 200 }); } catch (error) { return new Response("Something went wrong", { status: 500 }); } } };
